FULL_NAME, GOAL, EQUIPMENT, INJURIES, SCHEDULE = range(5)
GOAL_CHOICES = ("MMA / Combat", "Conditioning / Force", "Remise en forme")
EQUIPMENT_CHOICES = ("Poids du corps", "Haltères / Kettlebell", "Salle complète (Gym)")
SCHEDULE_CHOICES = ("Matin (8h-12h)", "Après-midi (14h-18h)", "Soir (18h-21h)")


def _database() -> Client:
    if supabase is None:
        raise RuntimeError("La connexion Supabase n'est pas configurée.")
    return supabase


def _first_row(response: Any) -> Optional[dict[str, Any]]:
    data = getattr(response, "data", None)
    return data[0] if data else None


def _athlete_name(athlete: dict[str, Any], fallback: str) -> str:
    """Compatibilité avec les schémas athletes V1 (first_name/last_name) et V2 (full_name)."""
    if athlete.get("full_name"):
        return athlete["full_name"]
    name = " ".join(part for part in (athlete.get("first_name"), athlete.get("last_name")) if part)
    return name or fallback


def _split_full_name(full_name: Optional[str]) -> tuple[str, str]:
    """Prépare le nom pour le schéma historique athletes (first_name/last_name)."""
    safe_name = (full_name or "").strip()
    parts = safe_name.split(maxsplit=1)
    return (parts[0] if parts else "", parts[1] if len(parts) > 1 else "")


def _upsert_athlete(user: Any, full_name: Optional[str] = None) -> dict[str, Any]:
    """Écrit dans le schéma V2, avec repli transparent vers le schéma historique."""
    existing = _first_row(_database().table("athletes").select("*").eq("telegram_id", str(user.id)).execute())
    
    clean_first_name = (user.first_name or "").strip()
    clean_last_name = (user.last_name or "").strip()
    clean_username = (user.username or "").strip()
    clean_full_name = (full_name or "").strip() if full_name else None

    data = {
        "telegram_id": str(user.id),
        "username": clean_username,
        "first_name": clean_first_name,
        "last_name": clean_last_name,
        "is_active": True,
        "code_id": f"TG-{user.id}"
    }
    
    if clean_full_name:
        data["full_name"] = clean_full_name
    
    if not existing:
        data.update({
            "status": "trial",
            "trial_sessions_left": 3,
            "subscription_tier": "trial"
        })
    
    try:
        response = _database().table("athletes").upsert(data, on_conflict="telegram_id").execute()
    except Exception as v2_error:
        logger.warning("Schéma athletes V2 indisponible, repli vers le schéma historique : %s", v2_error)
        first_name_split, last_name_split = _split_full_name(clean_full_name)
        response = _database().table("athletes").upsert(
            {
                "code_id": f"TG-{user.id}",
                "telegram_id": user.id,
                "first_name": first_name_split or clean_first_name,
                "last_name": last_name_split or clean_last_name,
                "tier": 1,
            },
            on_conflict="telegram_id",
        ).execute()
    athlete = _first_row(response)
    if athlete is None:
        athlete = _first_row(_database().table("athletes").select("*").eq("telegram_id", str(user.id)).execute())
    if athlete is None:
        raise RuntimeError("Supabase n'a pas renvoyé l'identifiant de l'athlète.")
    return athlete


def _upsert_profile(athlete_id: str, user: Any, data: dict[str, str]) -> None:
    """Écrit le profil V2, ou les champs équivalents du schéma historique."""
    try:
        _database().table("athlete_profiles").upsert(
            {
                "athlete_id": athlete_id,
                "goal": data["goal"],
                "default_equipment": data["equipment"],
                "injuries_history": data["injuries"],
                "preferred_schedule": data["schedule"],
                "language": user.language_code or "fr",
            },
            on_conflict="athlete_id",
        ).execute()
    except Exception as v2_error:
        logger.warning("Schéma athlete_profiles V2 indisponible, repli vers le schéma historique : %s", v2_error)
        injuries = (data["injuries"] or "").strip()
        _database().table("athlete_profiles").upsert(
            {
                "athlete_id": athlete_id,
                "goal": data["goal"],
                "available_equipment": [data["equipment"]],
                "injuries_and_constraints": [] if injuries.lower() in {"aucune", "aucun", "non", "none"} else [injuries],
            },
            on_conflict="athlete_id",
        ).execute()


async def start_onboarding(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    if not await _pre_process_user(update):
        return ConversationHandler.END
    user, message = update.effective_user, update.effective_message
    try:
        athlete = _first_row(_database().table("athletes").select("*").eq("telegram_id", str(user.id)).execute())
        profile = (
            _first_row(_database().table("athlete_profiles").select("athlete_id").eq("athlete_id", athlete["id"]).execute())
            if athlete
            else None
        )
    except Exception:
        logger.exception("Échec de la recherche de l'athlète %s", user.id)
        await message.reply_text("Je ne peux pas accéder à ton profil pour le moment. Réessaie dans quelques instants.")
        return ConversationHandler.END
    if athlete and profile:
        await message.reply_text(f"Content de te revoir {_athlete_name(athlete, user.first_name)} ! 🥋\nEnvoie-moi ton check-in vocal ou texte du jour quand tu es prêt.")
        return ConversationHandler.END
    context.user_data.clear()
    await message.reply_text("Bienvenue dans Amazonian Samourai Coaching ! 🥊\n\nJe vais créer ton profil en cinq questions rapides.\n\n1️⃣ Quel est ton nom et prénom ?")
    return FULL_NAME

async def finish_onboarding(update: Update, context: ContextTypes.DEFAULT_TYPE) -> int:
    message, user = update.effective_message, update.effective_user
    choice = message.text if message else ""
    if choice not in SCHEDULE_CHOICES:
        await message.reply_text("Choisis une option parmi les boutons proposés.")
        return SCHEDULE
    if user is None or message is None:
        return ConversationHandler.END
    try:
        athlete = _upsert_athlete(user, context.user_data["full_name"])
        _upsert_profile(
            athlete["id"],
            user,
            {
                "goal": context.user_data["goal"],
                "equipment": context.user_data["equipment"],
                "injuries": context.user_data["injuries"],
                "schedule": choice,
            },
        )
    except Exception as error:
        logger.exception("Échec de l'enregistrement de l'onboarding pour %s", user.id)
        diagnostic = str(error)
        if SUPABASE_KEY:
            diagnostic = diagnostic.replace(SUPABASE_KEY, "[clé masquée]")
        diagnostic = diagnostic.replace("\n", " ")[:350]
        await message.reply_text(
            "Je n'ai pas pu enregistrer ton profil.\n\n"
            f"Diagnostic Supabase : {diagnostic or 'erreur non détaillée'}",
            reply_markup=ReplyKeyboardRemove(),
        )
        return ConversationHandler.END
    context.user_data.clear()
    await message.reply_text("✅ Profil configuré !\n\nChaque jour, envoie-moi un vocal ou un texte sur ton sommeil, ton énergie, ton stress et tes douleurs.", reply_markup=ReplyKeyboardRemove())
    return ConversationHandler.END

async def _get_athlete_summary(athlete_id: str) -> str:
    """Génère un résumé coach pour un athlète donné."""
    try:
        # Récupère les données des 7 derniers jours
        checkins = _database().table("checkins").select("*").eq("athlete_id", athlete_id).order("created_at", desc=True).limit(7).execute()
        workouts = _database().table("workout_logs").select("*").eq("athlete_id", athlete_id).order("created_at", desc=True).limit(7).execute()
        debriefs = _database().table("debriefs").select("*").eq("athlete_id", athlete_id).order("created_at", desc=True).limit(7).execute()
        
        if not checkins.data and not workouts.data:
            return "Aucune donnée récente pour cet athlète."
            
        prompt = f"""Génère un rapport hebdomadaire structuré pour l'athlète {athlete_id}:

1. Tendance Readiness (7 derniers jours):
- Scores: {[c['readiness_score'] for c in checkins.data]}
- Statuts: {[c['readiness_status'] for c in checkins.data]}

2. Performances d'entraînement:
- RPE moyen: {sum(w['rpe_score'] for w in workouts.data)/len(workouts.data) if workouts.data else 'N/A'}
- Volume: {len(workouts.data)} séances
- Feedback: {[d['feedback'] for d in debriefs.data if d.get('feedback')]}

3. Points d'attention physique:
- Douleurs signalées: {[c['pain_location'] for c in checkins.data if c['pain_score'] >= 3]}
- Scores douleur: {[c['pain_score'] for c in checkins.data if 'pain_score' in c]}

4. Recommandations pour la semaine à venir:
"""
        return await GeminiService.generate_coach_summary(prompt)
    except Exception as e:
        logger.error(f"Erreur génération summary: {e}")
        return f"Erreur lors de la génération du résumé: {str(e)}"

async def handle_summary(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    """Commande admin pour obtenir un résumé d'athlète."""
    message = update.effective_message
    user = update.effective_user
    
    if not user or user.id not in ADMIN_TELEGRAM_IDS:
        await message.reply_text("Commande réservée aux coachs.")
        return
        
    args = (message.text or "").split()
    if len(args) < 2:
        await message.reply_text("Usage: /summary <telegram_id_ou_username>")
        return
        
    target = args[1].strip()
    try:
        # Recherche de l'athlète
        athlete = _first_row(_database().table("athletes")
            .select("*")
            .or_(f"telegram_id.eq.{target},username.eq.{target}")
            .execute())
            
        if not athlete:
            await message.reply_text("Athlète non trouvé.")
            return
            
        summary = await _get_athlete_summary(athlete["id"])
        await context.bot.send_message(
            chat_id=user.id,
            text=f"📊 Rapport Coach - {athlete.get('full_name') or athlete['username']} (Tier {athlete.get('tier', '1')})\n\n{summary}"
        )
    except Exception as e:
        logger.error(f"Erreur commande summary: {e}")
        await message.reply_text("Erreur lors de la génération du rapport.")

async def handle_vip_code(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    message = update.effective_message
    if not message or not message.text:
        return
        
    args = (message.text or "").split()
    if len(args) < 2:
        await message.reply_text("Utilisation : /code <votre_code_secret>")
        return
        
    code = args[1].strip().upper()
    user = update.effective_user
    if not user:
        return
        
    try:
        # Recherche du code promo actif
        promo_code = _first_row(_database().table("promo_codes")
            .select("*")
            .eq("code", code)
            .eq("is_active", True)
            .execute())
            
        if not promo_code:
            await message.reply_text("Code invalide ou expiré.")
            return
            
        if promo_code["times_used"] >= promo_code["max_uses"]:
            await message.reply_text("Ce code a atteint sa limite maximale d'utilisations.")
            return
            
        # Mise à jour du code promo
        _database().table("promo_codes").update({
            "times_used": promo_code["times_used"] + 1
        }).eq("id", promo_code["id"]).execute()
        
        # Mise à jour du statut de l'athlète
        _database().table("athletes").update({
            "status": "vip",
            "trial_sessions_left": 0,
            "subscription_tier": "vip",
            "used_promo_code": promo_code["code"]
        }).eq("telegram_id", str(user.id)).execute()
        
        await message.reply_text(
            "🎉 Accès VIP activé !\n\n"
            "Tu as maintenant un accès illimité à Amazonian Samourai Coaching.\n"
            "Ton livre et programme complet sont disponibles ici : [LINK]"
        )
    except Exception as e:
        logger.error(f"Erreur activation VIP : {e}")
        await message.reply_text("Erreur lors de l'activation. Contacte le coach.")


def calculate_module2_readiness(checkin: DailyCheckinInput) -> tuple[float, str, ReadinessResult]:
    """Calcule la readiness du Module 2 : la fatigue ne fait volontairement pas partie de la formule."""
    score = round(
        checkin.sleep_score * 0.35
        + checkin.energy_score * 0.35
        + (6 - checkin.stress_score) * 0.15
        + (6 - checkin.soreness_score) * 0.15,
        2,
    )
    if score >= 4.0:
        label, status, volume, rpe = "FORME_OPTIMALE", ReadinessStatus.GREEN, 1.0, None
    elif score >= 3.0:
        label, status, volume, rpe = "CHARGE_MODEREE", ReadinessStatus.ORANGE, 0.8, 7
    else:
        label, status, volume, rpe = "RECUPERATION_ACTIVE", ReadinessStatus.RED, 0.5, 4
    return score, label, ReadinessResult(
        score=score,
        status=status,
        details={"sleep": checkin.sleep_score, "energy": checkin.energy_score, "stress_inverse": 6 - checkin.stress_score, "pain_inverse": 6 - checkin.soreness_score},
        intensity_cap_rpe=rpe,
        volume_multiplier=volume,
        recommendation=label.replace("_", " "),
    )


def _pain_location(checkin: DailyCheckinInput) -> Optional[str]:
    return ", ".join(checkin.soreness_locations) if checkin.soreness_locations else None


def _is_debrief(text: str) -> bool:
    return bool(re.search(r"\b(rpe|débrief|debrief|séance validée|séance validee|ressenti)\b", text, re.IGNORECASE))


def _extract_rpe(text: str) -> Optional[int]:
    match = re.search(r"\brpe\s*[:=]?\s*(10|[1-9])(?:\s*/\s*10)?\b|\b(10|[1-9])\s*/\s*10\b", text, re.IGNORECASE)
    return int(match.group(1) or match.group(2)) if match else None

async def _registered_athlete(telegram_id: int) -> Optional[dict[str, Any]]:
    try:
        # Requête simplifiée pour récupérer uniquement les champs nécessaires
        athlete = _first_row(_database().table("athletes")
            .select("id,telegram_id,first_name,last_name")
            .eq("telegram_id", str(telegram_id))
            .execute())
        
        if not athlete:
            logger.warning("Aucun athlète trouvé avec telegram_id=%s", telegram_id)
            return None
            
        # Vérification stricte du format UUID
        if not isinstance(athlete.get("id"), str) or len(athlete["id"]) < 16:
            logger.error("Format d'ID athlète invalide : %s", athlete.get("id"))
            return None
            
        return athlete
    except Exception as e:
        logger.exception("Échec de la recherche de l'athlète %s : %s", telegram_id, str(e))
        return None


async def _athlete_profile(athlete: dict[str, Any]) -> AthleteProfile:
    profile = _first_row(_database().table("athlete_profiles").select("*").eq("athlete_id", athlete["id"]).execute()) or {}
    equipment = profile.get("default_equipment") or profile.get("available_equipment") or []
    injuries = profile.get("injuries_history") or profile.get("injuries_and_constraints") or []
    if isinstance(equipment, str):
        equipment = [equipment]
    if isinstance(injuries, str):
        injuries = [] if (injuries or "").strip().lower() in {"aucune", "aucun", "non", "none"} else [injuries]
    return AthleteProfile(
        athlete_id=athlete["id"], telegram_id=int(athlete["telegram_id"]),
        first_name=athlete.get("first_name") or _athlete_name(athlete, "Athlète").split()[0],
        tier=athlete.get("tier", 1), goal=profile.get("goal"),
        available_equipment=equipment, injuries_and_constraints=injuries,
    )


async def _store_checkin(athlete_id: str, checkin: Any, score: float, label: str, source: str = "text"):
    # Gestion des types Pydantic et dictionnaire
    if hasattr(checkin, "model_dump"):
        checkin_dict = checkin.model_dump()
    elif hasattr(checkin, "dict"):
        checkin_dict = checkin.dict()
    else:
        checkin_dict = dict(checkin)
    
    payload = {
        "athlete_id": athlete_id,
        "sleep_score": checkin_dict.get("sleep_score"),
        "energy_score": checkin_dict.get("energy_score"),
        "stress_score": checkin_dict.get("stress_score"),
        "pain_score": checkin_dict.get("pain_score") or checkin_dict.get("soreness_score"),
        "pain_location": checkin_dict.get("pain_location") or (
            ", ".join(checkin_dict.get("soreness_locations", [])) if checkin_dict.get("soreness_locations") else None
        ),
        "readiness_score": score,
        "readiness_status": label,
        "raw_text": checkin_dict.get("raw_text", "")
    }
    logger.info(f"Insertion check-in Supabase pour athlete_id={athlete_id} : {payload}")
    try:
        response = _database().table("checkins").insert(payload).execute()
        return response
    except Exception as e:
        logger.error(f"Erreur Supabase lors du checkin : {e}", exc_info=True)
        raise RuntimeError("Échec de l'enregistrement du check-in") from e

async def _store_workout_log(athlete_id: str, rpe_score: int, feedback_text: str) -> None:
    # Enregistrement du log
    _database().table("workout_logs").insert({
        "athlete_id": athlete_id,
        "rpe_score": rpe_score,
        "feedback_text": feedback_text,
        "log_type": "workout_completion"
    }).execute()
    
    # Gestion des séances d'essai
    athlete = _first_row(_database().table("athletes").select("*").eq("id", athlete_id).execute())
    if athlete and athlete.get("status") == "trial":
        sessions_left = athlete.get("trial_sessions_left", 3) - 1
        update_data = {"trial_sessions_left": sessions_left}
        if sessions_left <= 0:
            update_data["status"] = "expired"
        _database().table("athletes").update(update_data).eq("id", athlete_id).execute()


async def _process_debrief(update: Update, text: str) -> None:
    message, user = update.effective_message, update.effective_user
    if message is None or user is None:
        return
    athlete = await _registered_athlete(user.id)
    if athlete is None:
        await message.reply_text("Ton profil n'est pas encore créé. Tape /start pour commencer.")
        return
    try:
        await _store_debrief(athlete["id"], text, _extract_rpe(text))
    except Exception:
        logger.exception("Échec de l'enregistrement du débrief pour %s", athlete["id"])
        await message.reply_text("Je n'ai pas pu enregistrer ce débrief. Réessaie dans un instant.")
        return
    await message.reply_text("✅ Débrief enregistré. Merci — la prochaine séance tiendra compte de ton retour.")


async def _check_access(user_id: int) -> bool:
    athlete = await _registered_athlete(user_id)
    if not athlete:
        return False
        
    status = athlete.get("status")
    sessions_left = athlete.get("trial_sessions_left", 0)
    
    if status in ("vip", "active"):
        return True
    if status == "expired" or (status == "trial" and sessions_left <= 0):
        return False
    return True

def calculate_module2_readiness(checkin: DailyCheckinInput) -> tuple[float, str, ReadinessResult]:
    """Calcule la readiness du Module 2 : la fatigue ne fait volontairement pas partie de la formule."""
    score = round(
        checkin.sleep_score * 0.35
        + checkin.energy_score * 0.35
        + (6 - checkin.stress_score) * 0.15
        + (6 - checkin.soreness_score) * 0.15,
        2,
    )
    if score >= 4.0:
        label, status, volume, rpe = "FORME_OPTIMALE", ReadinessStatus.GREEN, 1.0, None
    elif score >= 3.0:
        label, status, volume, rpe = "CHARGE_MODEREE", ReadinessStatus.ORANGE, 0.8, 7
    else:
        label, status, volume, rpe = "RECUPERATION_ACTIVE", ReadinessStatus.RED, 0.5, 4
    return score, label, ReadinessResult(
        score=score,
        status=status,
        details={"sleep": checkin.sleep_score, "energy": checkin.energy_score, "stress_inverse": 6 - checkin.stress_score, "pain_inverse": 6 - checkin.soreness_score},
        intensity_cap_rpe=rpe,
        volume_multiplier=volume,
        recommendation=label.replace("_", " "),
    )

def create_telegram_application() -> Application:
    token = os.getenv("TELEGRAM_BOT_TOKEN")
    if not token:
        raise ValueError("TELEGRAM_BOT_TOKEN manquant dans l'environnement.")
    application = ApplicationBuilder().token(token).build()

    )

    application.add_handler(conv_handler)
    application.add_handler(CommandHandler("code", code_command))
    application.add_handler(CommandHandler("summary", summary_command))
